cmake_minimum_required(VERSION 3.13)

if(NOT DEFINED ENV{PICO_SDK_PATH})
    set(ENV{PICO_SDK_PATH} "/home/tane/dev/PicoW/pico-sdk")
endif()
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
option(PLAYER_CYW43_DIAG "Build with isolated CYW43 loss instrumentation" OFF)
option(PLAYER_LCD_GAMMA_LGFX "Use the adopted ST7735S E0/E1 gamma pair" ON)
if(PLAYER_CYW43_DIAG)
    set(PICO_CYW43_DRIVER_PATH "${CMAKE_CURRENT_LIST_DIR}/diagnostics/pico/cyw43-driver")
endif()
project(pico_w_jpeg_player C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()

add_executable(pico_w_jpeg_player pico/pico_player.c pico/jpeg_decode.c pico/third_party/tjpgd/lgfx_tjpgd.c)
target_include_directories(pico_w_jpeg_player PRIVATE
    ${CMAKE_CURRENT_LIST_DIR}
    ${CMAKE_CURRENT_LIST_DIR}/pico)
if(PLAYER_CYW43_DIAG)
    target_sources(pico_w_jpeg_player PRIVATE diagnostics/pico/cyw43_diag.c)
    target_include_directories(pico_w_jpeg_player PRIVATE ${CMAKE_CURRENT_LIST_DIR}/diagnostics/pico)
    target_compile_definitions(pico_w_jpeg_player PRIVATE PLAYER_CYW43_DIAG=1)
endif()
if(PLAYER_LCD_GAMMA_LGFX)
    target_compile_definitions(pico_w_jpeg_player PRIVATE PLAYER_LCD_GAMMA_LGFX=1)
endif()
set(DISPLAY_PROFILE 0 CACHE STRING "0=existing breadboard SPI0, 1=proposed SPI1 layout")
target_compile_definitions(pico_w_jpeg_player PRIVATE DISPLAY_PROFILE=${DISPLAY_PROFILE})
set(PLAYER_STATIC_IP "192.168.0.173" CACHE STRING "Pico W station IPv4 address")
set(PLAYER_GATEWAY "192.168.0.1" CACHE STRING "LAN gateway")
set(PLAYER_NETMASK "255.255.255.0" CACHE STRING "LAN netmask")
set(PLAYER_DNS "192.168.0.1" CACHE STRING "LAN DNS")
pico_configure_ip4_address(pico_w_jpeg_player PRIVATE CYW43_DEFAULT_IP_STA_ADDRESS "${PLAYER_STATIC_IP}")
pico_configure_ip4_address(pico_w_jpeg_player PRIVATE CYW43_DEFAULT_IP_STA_GATEWAY "${PLAYER_GATEWAY}")
pico_configure_ip4_address(pico_w_jpeg_player PRIVATE CYW43_DEFAULT_IP_MASK "${PLAYER_NETMASK}")
pico_configure_ip4_address(pico_w_jpeg_player PRIVATE CYW43_DEFAULT_IP_DNS "${PLAYER_DNS}")
target_link_libraries(pico_w_jpeg_player pico_stdlib hardware_spi pico_cyw43_arch_lwip_threadsafe_background)
pico_enable_stdio_usb(pico_w_jpeg_player 0)
pico_enable_stdio_uart(pico_w_jpeg_player 0)
pico_add_extra_outputs(pico_w_jpeg_player)
